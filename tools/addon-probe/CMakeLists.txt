cmake_minimum_required(VERSION 3.27)
project(fcitx5-windows-addon-probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Fcitx5Core 5.1.13 REQUIRED)
add_executable(addon_probe main.cpp ../../src/windowsplugins.cpp)
target_link_libraries(addon_probe PRIVATE Fcitx5::Core shell32 ole32)
target_compile_options(addon_probe PRIVATE -fexperimental-library)
target_link_options(addon_probe PRIVATE -municode)
install(TARGETS addon_probe RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_windowsplugins test_windowsplugins.cpp ../../src/windowsplugins.cpp)
    target_link_libraries(test_windowsplugins PRIVATE Fcitx5::Core shell32 ole32)
    target_compile_options(test_windowsplugins PRIVATE -fexperimental-library)
    add_test(NAME windowsplugins COMMAND test_windowsplugins)
    set_tests_properties(windowsplugins PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${FCITX5_WINDOWS_SDK}/bin")
endif()
